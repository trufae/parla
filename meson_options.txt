option('webxdc',
  type: 'boolean',
  value: true,
  description: 'Run Webxdc (Delta Chat mini-app) attachments in an embedded web view. Enabled by default; pass -Dwebxdc=false to disable. Uses WebKitGTK on Linux, system WebKit on macOS, and system Edge WebView2 on Windows; see docs/webxdc.md.')
option('webview2_sdk',
  type: 'string',
  value: '',
  description: 'Windows WebView2 SDK package root (the directory containing build/native/include and the architecture-specific WebView2Loader.dll). Required for enabled Webxdc Windows builds.')
option('rpc_server_path',
  type: 'string',
  value: '',
  description: 'Absolute path to a fixed deltachat-rpc-server (Delta Chat jsonrpc server) binary. When set, Parla uses this binary as the Custom server and disables engine auto-updates.')
option('bundle_icons',
  type: 'boolean',
  value: false,
  description: 'Compile the Adwaita symbolic icons Parla uses into the binary as hicolor fallbacks. Off by default: at runtime Parla inherits them from the installed adwaita-icon-theme (a GTK4 dependency) when the active icon theme lacks them. Enable for builds that cannot rely on an installed Adwaita.')
option('a11y',
  type: 'boolean',
  value: true,
  description: 'Expose the interface to screen readers: accessible names and states on the widgets Parla builds by hand, plus the GTK assistive-technology backend (AT-SPI, AccessKit) at startup. Enabled by default; pass -Da11y=false for a build that starts with GTK_A11Y=none and carries no accessibility code (see docs/accessibility.md).')
option('sailfish',
  type: 'boolean',
  value: false,
  description: 'Build the Sailfish OS port: disable GTK AT-SPI at startup and use phone-oriented scaling, layout and window chrome.')
