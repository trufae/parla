option('webxdc',
  type: 'boolean',
  value: false,
  description: 'Experimental: run Webxdc (Delta Chat mini-app) attachments in an embedded web view. Uses WebKitGTK on Linux, system WebKit on macOS, and system Edge WebView2 on Windows; see docs/webxdc.md.')
option('webview2_sdk',
  type: 'string',
  value: '',
  description: 'Windows WebView2 SDK package root (the directory containing build/native/include and the architecture-specific WebView2Loader.dll). Required only for -Dwebxdc=true Windows builds.')
option('rpc_server_path',
  type: 'string',
  value: '',
  description: 'Absolute path to a fixed deltachat-rpc-server (Delta Chat jsonrpc server) binary. When set, Parla uses this binary as the Custom server and disables engine auto-updates.')
option('bundle_icons',
  type: 'boolean',
  value: true,
  description: 'Compile the Adwaita symbolic icons Parla uses into the binary as hicolor fallbacks, so the UI renders on desktops whose icon theme lacks them (XFCE, LXQt, ...). Disable to rely on an installed adwaita-icon-theme instead.')
